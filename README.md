        // Xử lý Slider 2 Banner tự trượt
        let currentBannerIdx = 0;
        const bannerSlides = () => document.querySelectorAll('.banner-slide');

        function showBanner(index) {
            const slides = bannerSlides();
            if (!slides.length) return;
            currentBannerIdx = (index + slides.length) % slides.length;
            slides.forEach((sl, i) => {
                if (i === currentBannerIdx) {
                    sl.classList.remove('hidden');
                } else {
                    sl.classList.add('hidden');
                }
            });
            const dot0 = document.getElementById('bannerDot0');
            const dot1 = document.getElementById('bannerDot1');
            if (dot0 && dot1) {
                dot0.className = currentBannerIdx === 0 ? 'w-6 h-2 rounded-full bg-amber-400 transition-all' : 'w-2.5 h-2 rounded-full bg-white/50 hover:bg-white transition-all';
                dot1.className = currentBannerIdx === 1 ? 'w-6 h-2 rounded-full bg-amber-400 transition-all' : 'w-2.5 h-2 rounded-full bg-white/50 hover:bg-white transition-all';
            }
        }

        function nextBanner() {
            showBanner(currentBannerIdx + 1);
        }
        function prevBanner() {
            showBanner(currentBannerIdx - 1);
        }
        function goToBanner(idx) {
            showBanner(idx);
        }
